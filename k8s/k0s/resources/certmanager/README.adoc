= Cert Manager

Setup credential

[source,bash]
----
cat <<EOF | kubectl apply -f -
---
apiVersion: v1
kind: Secret
metadata:
  name: infomaniak-api-credentials
  namespace: cert-manager
type: Opaque
data:
  api-token: $(echo -n $INFOMANIAK_TOKEN|base64 -w0)
EOF
----

Generate secret
[source,bash]
----
apiVersion: cert-manager.io/v1
kind: Certificate
metadata:
  name: test-gssn-xyz
spec:
  secretName: test-gssn-xyz-tls
  issuerRef:
    name: letsencrypt-staging
    kind: ClusterIssuer
  dnsNames:
  - test.gssn.xyz
----
